# Full path to your repository on the Droplet
REPO_DIR=/home/yourusername/The_Regulated_Friction_Project

# Your Streamlit dashboard URL (find at https://share.streamlit.io/)
DASHBOARD_URL=https://your-app-name.streamlit.app

# ── OPTIONAL CONFIGURATION ──────────────────────────────────────────────────

# Shell and PATH
SHELL=/bin/bash
PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin

# Python executable (verify with: which python3)
PYTHON=/usr/bin/python3

# Email for cron failure notifications (optional - comment out to disable)
# MAILTO=your-email@example.com

# ── CRON SCHEDULE ───────────────────────────────────────────────────────────
# Format: minute hour day month weekday command
#
# Examples:
#   0 2 * * *     = Every day at 2:00 AM
#   */6 * * * *   = Every 6 hours
#   0 9 */6 * *   = Every 6 days at 9:00 AM
# ============================================================================

# ┌─────────────────────────────────────────────────────────────────────────┐
# │ DAILY SPIDER RUN - Federal Register Executive Orders                    │
# ├─────────────────────────────────────────────────────────────────────────┤
# │ Scrapes the last 7 days of Executive Orders, Proclamations, and Notices │
# │ Saves output to: federal_register/Spider Output Files/                  │
# │ Default: 2:00 AM UTC daily                                              │
# └─────────────────────────────────────────────────────────────────────────┘
0 2 * * * cd $REPO_DIR && $PYTHON -m scrapy crawl federal_register_eo -o "federal_register/Spider Output Files/items_federal_register_eo_1.json" >> /var/log/spider_run.log 2>&1

# ┌─────────────────────────────────────────────────────────────────────────┐
# │ PUSH DATA TO GITHUB                                                     │
# ├─────────────────────────────────────────────────────────────────────────┤
# │ Commits and pushes spider output to GitHub repository                   │
# │ Default: 2:15 AM UTC daily (15 min after spider completes)              │
# └─────────────────────────────────────────────────────────────────────────┘
15 2 * * * $REPO_DIR/push_spider_data.sh

# ┌─────────────────────────────────────────────────────────────────────────┐
# │ KEEP STREAMLIT DASHBOARD AWAKE                                          │
# ├─────────────────────────────────────────────────────────────────────────┤
# │ Pings dashboard to prevent it from sleeping after 7 days of inactivity  │
# │ Default: 9:00 AM UTC every 6 days                                       │
# └─────────────────────────────────────────────────────────────────────────┘
0 9 */6 * * /usr/bin/curl -s -o /dev/null -w "%{http_code}\n" "$DASHBOARD_URL" >> /var/log/dashboard_ping.log 2>&1
